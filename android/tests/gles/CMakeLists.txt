cmake_minimum_required(VERSION 3.22)
project(ConkerGLESProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CONKER_RT64 OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/android/native/gles" "${CMAKE_CURRENT_BINARY_DIR}/gles")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_library(CONKER_EGL_LIBRARY NAMES EGL libEGL.so.1 REQUIRED)
add_executable(gles_regression gles_regression.cpp)
target_include_directories(gles_regression PRIVATE "${ROOT}/android/native")
target_link_libraries(gles_regression PRIVATE conker_gles_core ${CONKER_EGL_LIBRARY})
option(CONKER_GLES_ROM_PROBE "Build the private real-game probe (requires user ROM-generated sources)" OFF)
if(CONKER_GLES_ROM_PROBE)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/prepare_probe.py")
    add_subdirectory("${ROOT}/host" "${CMAKE_CURRENT_BINARY_DIR}/host")
    get_target_property(GAME_SOURCES RecompiledFuncs SOURCES)
    set(HOOK_SOURCES)
    foreach(SYMBOL func_15123508 func_1501B22C)
        execute_process(COMMAND ${Python3_EXECUTABLE} "${ROOT}/android/tools/game_sources.py" "${ROOT}/RecompiledFuncs" ${SYMBOL}
            OUTPUT_VARIABLE HOOK_SOURCE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        list(APPEND HOOK_SOURCES "${HOOK_SOURCE}")
    endforeach()
    list(REMOVE_DUPLICATES HOOK_SOURCES)
    foreach(HOOK_SOURCE IN LISTS HOOK_SOURCES)
        get_filename_component(HOOK_NAME "${HOOK_SOURCE}" NAME)
        execute_process(COMMAND ${Python3_EXECUTABLE} "${ROOT}/android/tools/prepare_camera_hook.py"
            "${HOOK_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/hook_${HOOK_NAME}" COMMAND_ERROR_IS_FATAL ANY)
        list(FILTER GAME_SOURCES EXCLUDE REGEX "${HOOK_NAME}$")
        list(APPEND GAME_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/hook_${HOOK_NAME}")
    endforeach()
    set_property(TARGET RecompiledFuncs PROPERTY SOURCES "${GAME_SOURCES}")
    execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/prepare_probe.py"
        "${ROOT}/host/src/main.cpp" "${CMAKE_CURRENT_BINARY_DIR}/probe_main.cpp" COMMAND_ERROR_IS_FATAL ANY)
    get_target_property(SOURCES ConkerRecomp SOURCES)
    list(REMOVE_ITEM SOURCES "src/main.cpp" "src/null_renderer.cpp")
    set_property(TARGET ConkerRecomp PROPERTY SOURCES "${SOURCES};${CMAKE_CURRENT_BINARY_DIR}/probe_main.cpp;${CMAKE_CURRENT_SOURCE_DIR}/gles_probe.cpp;${ROOT}/android/native/mobile_camera.cpp;${ROOT}/android/native/mobile_frustum.cpp")
    target_include_directories(ConkerRecomp PRIVATE "${ROOT}/host/src" "${ROOT}/android/native")
    target_link_libraries(ConkerRecomp PRIVATE conker_gles_core ${CONKER_EGL_LIBRARY})

endif()
